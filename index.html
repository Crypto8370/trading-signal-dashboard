async function updateSignal() { const signalText = document.getElementById('signal-text'); const signalBox = document.getElementById('signal-box'); const confVal = document.getElementById('conf-val'); try { const response = await fetch('https://api.binance.com/api/v3/ticker/24hr?symbol=BTCUSDT'); const data = await response.json(); // Definisi variabel dengan huruf kapital yang konsisten const priceChange = parseFloat(data.priceChangePercent); const lastPrice = parseFloat(data.lastPrice).toFixed(2); if (priceChange > 0.1) { signalText.innerText = "BUY"; signalText.className = "text-5xl font-black my-4 text-green-500 animate-pulse"; signalBox.style.borderTopColor = "#22c55e"; confVal.innerText = (70 + Math.random() * 20).toFixed(0) + "%"; } else if (priceChange < -0.1) { signalText.innerText = "SELL"; signalText.className = "text-5xl font-black my-4 text-red-500 animate-pulse"; signalBox.style.borderTopColor = "#ef4444"; confVal.innerText = (70 + Math.random() * 20).toFixed(0) + "%"; } else { signalText.innerText = "NEUTRAL"; signalText.className = "text-5xl font-black my-4 text-gray-400"; signalBox.style.borderTopColor = "#6b7280"; confVal.innerText = "Stable"; } // Perbaikan pemanggilan ID dan Variabel document.getElementById('entry-price').innerText = lastPrice; document.getElementById('tp-price').innerText = (parseFloat(lastPrice) * 1.01).toFixed(2); document.getElementById('sl-price').innerText = (parseFloat(lastPrice) * 0.995).toFixed(2); } catch (error) { console.error("Error fetching data:", error); } }
